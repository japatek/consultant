generator client {
  provider = "prisma-client"
  output   = "../src/generated/prisma"
}

datasource db {
  provider = "postgresql"
}

model User {
  id            String    @id @default(cuid())
  name          String?
  email         String    @unique
  emailVerified DateTime?
  image         String?
  lastActive    DateTime? @default(now())
  joinedDate    DateTime?

  // Application role — persisted in the JWT for fast middleware checks
  role          String    @default("USER")

  // Auth.js relations
  accounts        Account[]
  sessions        Session[]

  createdAt       DateTime  @default(now())
  updatedAt       DateTime  @updatedAt
  emailChangeLogs EmailChangeLog[]

  licenses        License[]
  articles        Article[]

  // ==========================================================
  // --- Relasi Quest (Misi) / Sertifikasi ---
  // ==========================================================
  authoredQuests   Quest[]              @relation("QuestAuthor")
  questCompletions QuestCompletion[]
  certifications   UserCertification[]
  progress         UserProgress?

  // ==========================================================
  // --- Langganan (Subscription) / Penagihan (Billing) ---
  // ==========================================================
  // Tier yang didenormalisasi — disimpan di User untuk pembacaan cepat 
  // tanpa join. Diperbarui oleh webhook Midtrans.
  currentTier    String    @default("FREE") // "FREE" | slug dari PricingPlan
  tierExpiresAt  DateTime?
  subscriptions  Subscription[]

  @@map("users")
}

model EmailChangeLog {
  id        String   @id @default(cuid())
  userId    String
  oldEmail  String
  newEmail  String
  createdAt DateTime @default(now())

  // Relationship back to your User model
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@index([userId, createdAt])
}

model Account {
  userId            String
  type              String
  provider          String
  providerAccountId String

  // OAuth token fields — Text to avoid 255-char truncation
  refresh_token     String?   @db.Text
  access_token      String?   @db.Text
  expires_at        Int?
  token_type        String?
  scope             String?
  id_token          String?   @db.Text
  session_state     String?

  createdAt         DateTime  @default(now())
  updatedAt         DateTime  @updatedAt

  user              User      @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@id([provider, providerAccountId])
  @@map("accounts")
}

// Session model is required by PrismaAdapter even when using JWT strategy.
// Auth.js uses it internally for account linking; it does NOT store JWT sessions.
model Session {
  sessionToken  String    @unique
  userId        String
  expires       DateTime

  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt

  user          User      @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@map("sessions")
}

// Shared table for:
//   1. Magic-link tokens      (identifier = email address)
//   2. Change-email tokens    (identifier = "change-email:{userId}:{newEmail}")
//
// Auth.js v5 automatically SHA-256 hashes tokens before calling
// adapter.createVerificationToken, so the `token` column stores only hashes.
model VerificationToken {
  identifier    String
  token         String    @unique
  expires       DateTime

  @@id([identifier, token])
  @@map("verification_tokens")
}

  exampleResult String   @db.Text

model License {

  isActive   Boolean  @default(false)

  createdAt  DateTime @default(now())

model Article {
  id          String   @id @default(cuid())
  slug        String   @unique

  // PERBAIKAN: Dibuat opsional (?) agar admin hardcoded bisa membuat artikel
  // tanpa memicu error "Foreign key constraint violated"
  authorId    String?
  author      User?    @relation(fields: [authorId], references: [id], onDelete: Cascade)

  // Bahasa Inggris (Default)
  title       String
  desc        String
  content     String   @db.Text
  imageUrl    String?  // Untuk gambar thumbnail
  
  // Bahasa Indonesia
  id_title    String?
  id_desc     String?
  id_content  String?  @db.Text
  
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  @@index([authorId])
}

enum AnswerType {
  MULTIPLE_CHOICE
  TEXT_INPUT
  FILE_UPLOAD
}

model Quest {
  id           String   @id @default(cuid())
  slug         String   @unique
  title        String
  difficulty   String   // "Beginner" | "Intermediate" | "Advanced" — lihat src/types/quest.ts
  category     String   // "2D Drafting" | "3D Modeling" | "Assembly" | ... — lihat src/types/quest.ts

  // Ganti dengan kolom JSON/AST jika Anda nantinya menghubungkan editor teks kaya yang terstruktur
  // (Tiptap, Lexical, ...). Untuk saat ini menggunakan teks biasa/markdown.
  description  String   @db.Text
  instructions String   @db.Text

  answerType   AnswerType
  // Bentuknya bergantung pada answerType — dipaksakan (enforced) oleh Zod saat penulisan (src/types/quest.ts),
  // bukan oleh Postgres:
  //   MULTIPLE_CHOICE -> { type, options: string[], correctIndex: number }
  //   TEXT_INPUT        -> { type, expectedValue: string, caseSensitive?, tolerance? }
  //   FILE_UPLOAD      -> { type, allowedExtensions: string[], maxSizeMB: number }
  answerConfig Json

  points       Int      @default(10)
  isPublished  Boolean  @default(false)

  authorId     String
  author       User     @relation("QuestAuthor", fields: [authorId], references: [id], onDelete: Cascade)

  media          QuestMedia[]
  completions    QuestCompletion[]
  certifications CertificationQuest[]

  createdAt    DateTime @default(now())
  updatedAt    DateTime @updatedAt

  @@index([category])
  @@index([authorId])
  @@map("quests")
}

model QuestMedia {
  id       String @id @default(cuid())
  questId  String
  quest    Quest  @relation(fields: [questId], references: [id], onDelete: Cascade)
  type     String // "IMAGE" | "VIDEO" | "PDF"
  url      String
  fileName String

  createdAt DateTime @default(now())

  @@index([questId])
  @@map("quest_media")
}

model QuestCompletion {
  id      String @id @default(cuid())
  userId  String
  user    User   @relation(fields: [userId], references: [id], onDelete: Cascade)
  questId String
  quest   Quest  @relation(fields: [questId], references: [id], onDelete: Cascade)

  submittedAnswer Json    // bentuknya (shape) menyesuaikan answerConfig dari Quest untuk answerType yang sama
  isCorrect       Boolean
  pointsEarned    Int     @default(0)
  attempt         Int     @default(1)

  createdAt DateTime @default(now())

  @@index([userId, questId])
  @@map("quest_completions")
}

model Certification {
  id          String  @id @default(cuid())
  slug        String  @unique
  title       String  // contoh: "Persiapan Certified SolidWorks Associate"
  description String  @db.Text
  badgeUrl    String?

  requiredQuests CertificationQuest[]
  unlockedBy     UserCertification[]

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@map("certifications")
}

// Tabel join eksplisit (bukan m-n implisit) agar jika nanti ada kolom "required order" (urutan yang diwajibkan)
// atau "weight per quest" (bobot per quest) memiliki tempat untuk ditambahkan.
model CertificationQuest {
  certificationId String
  certification   Certification @relation(fields: [certificationId], references: [id], onDelete: Cascade)
  questId         String
  quest           Quest         @relation(fields: [questId], references: [id], onDelete: Cascade)

  @@id([certificationId, questId])
  @@map("certification_quests")
}

model UserCertification {
  id              String        @id @default(cuid())
  userId          String
  user            User          @relation(fields: [userId], references: [id], onDelete: Cascade)
  certificationId String
  certification   Certification @relation(fields: [certificationId], references: [id], onDelete: Cascade)

  // Kode publik yang dicetak pada sertifikat agar siapa pun dapat memverifikasi keasliannya
  verificationCode String   @unique @default(cuid())
  certificateUrl   String?  // diisi setelah PDF/gambar dibuat + diunggah
  unlockedAt       DateTime @default(now())

  @@unique([userId, certificationId])
  @@map("user_certifications")
}

// Profil gamifikasi yang didenormalisasi — satu baris per pengguna, lebih ringan/cepat dibaca saat
// setiap pemuatan dashboard dibandingkan harus mengagregasi QuestCompletion setiap saat.
model UserProgress {
  userId String @id
  user   User   @relation(fields: [userId], references: [id], onDelete: Cascade)

  totalPoints       Int       @default(0)
  questsCompleted   Int       @default(0)
  currentStreakDays Int       @default(0)
  lastActivityAt    DateTime?

  updatedAt DateTime @updatedAt

  @@map("user_progress")
}

enum PlanInterval {
  WEEKLY
  MONTHLY
  QUARTERLY
}

enum SubscriptionStatus {
  PENDING
  ACTIVE
  EXPIRED
  CANCELLED
}

model PricingPlan {
  id           String       @id @default(cuid())
  slug         String       @unique
  name         String       // "Mingguan", "Bulanan", "3 Bulan"
  interval     PlanInterval
  durationDays Int
  priceIDR     Int
  priceUSD     Int
  features     Json         // string[] — daftar poin (bullet list) yang ditampilkan pada kartu harga
  isActive     Boolean      @default(true)
  sortOrder    Int          @default(0)

  discounts     Discount[]
  subscriptions Subscription[]

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@map("pricing_plans")
}

model Discount {
  id             String       @id @default(cuid())
  code           String       @unique
  percentOff     Int?         // 0-100, pada praktiknya saling eksklusif (tidak bisa digunakan bersamaan) dengan amountOffIDR
  amountOffIDR   Int?
  planId         String?      // null = berlaku untuk setiap paket (plan)
  plan           PricingPlan? @relation(fields: [planId], references: [id], onDelete: Cascade)
  validFrom      DateTime     @default(now())
  validUntil     DateTime?
  maxRedemptions Int?
  timesRedeemed  Int          @default(0)
  isActive       Boolean      @default(true)

  createdAt DateTime @default(now())

  @@index([code])
  @@map("discounts")
}

model Subscription {
  id      String @id @default(cuid())
  userId  String
  user    User   @relation(fields: [userId], references: [id], onDelete: Cascade)
  planId  String
  // Sengaja tidak menggunakan onDelete: Cascade di sini — menghapus paket (plan) tidak boleh
  // menghapus riwayat catatan pembayaran. Nonaktifkan paket menggunakan `isActive`, jangan dihapus.
  plan    PricingPlan @relation(fields: [planId], references: [id])

  status  SubscriptionStatus @default(PENDING)
  startDate DateTime?
  endDate   DateTime?

  midtransOrderId       String  @unique
  midtransTransactionId String?
  amountPaidIDR         Int
  discountCodeUsed      String?

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@index([userId])
  @@index([midtransOrderId])
  @@map("subscriptions")
}
